class Solution {
public:
    int search(vector<int>& nums, int target) {
        int lo=0;
        int hi=nums.size()-1;
        int x=-1;
        while(lo<=hi){
            int mid=lo+hi;
            mid/=2;
            if(nums[mid]==target){
                x=mid;
                break;
            }
            else if(nums[mid]>target){
                hi=mid-1;
            }
            else{
                lo=mid+1;
            }
        }
        return x;   
    }
};
